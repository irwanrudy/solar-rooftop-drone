## Review Focus

Hal yang tidak dinyatakan spec tetapi paling mungkin mengganggu pengguna, beserta tes yang menguncinya:

1. **AOI sebagian di luar data valid ortofoto** (piksel hitam): segmen hitam tidak boleh jadi kandidat atap. Tes di Task 2.
2. **Nol denah Overture di AOI** (cakupan bolong): ringkasan tidak crash dan melaporkan nol. Tes di Task 1.
3. **Nol kandidat SAM setelah filter**: keluaran tetap ditulis (GeoJSON kosong valid) dan ada peringatan. Tes di Task 6.
4. **Denah Overture tidak valid atau MultiPolygon** (bow-tie): IoU tidak crash. Tes di Task 1.
5. **Satu segmen menempel di dua bangunan** (dinding bersama): hanya ditetapkan ke yang tumpang tindihnya terbesar, tanpa hitung ganda. Tes di Task 1.
6. **CUDA tidak tersedia**: jatuh ke CPU dengan peringatan, bukan crash. Tes di Task 4.

```python
"""Pencocokan poligon hasil SAM dengan denah Overture, dan metrik pergeseran.

Fungsi murni (tanpa torch). Satu atap sering dipecah SAM menjadi beberapa
segmen, jadi pencocokan dilakukan many-to-one: segmen ditetapkan ke denah
dengan tumpang tindih terbesar, segmen satu denah digabung, lalu IoU dihitung.
Indeks `overture` harus numerik (hasil `gpd.read_file`).
"""

from __future__ import annotations

from dataclasses import dataclass

import geopandas as gpd
import numpy as np
import pandas as pd
from shapely.geometry.base import BaseGeometry
from shapely.ops import unary_union


def _valid_gdf(gdf: gpd.GeoDataFrame) -> gpd.GeoDataFrame:
    """Salinan dengan geometri diperbaiki (make_valid) agar operasi overlay aman."""
    out = gdf.copy()
    if len(out):
        out["geometry"] = out.geometry.make_valid()
    return out


def iou(a: BaseGeometry, b: BaseGeometry) -> float:
    """Intersection-over-Union dua geometri (0 jika gabungannya kosong)."""
    from shapely.validation import make_valid

    a = a if a.is_valid else make_valid(a)
    b = b if b.is_valid else make_valid(b)
    union = a.union(b).area
    return float(a.intersection(b).area / union) if union else 0.0


def assign_segments(
    overture: gpd.GeoDataFrame, segments: gpd.GeoDataFrame, min_overlap: float = 0.5
) -> pd.Series:
    """Tetapkan tiap segmen ke denah Overture dengan irisan terbesar.

    Segmen ditetapkan bila `luas(irisan) / luas(segmen) >= min_overlap`.
    Mengembalikan Series (indeks = indeks segmen) berisi indeks denah, atau NaN.
    """
    out = pd.Series(np.nan, index=segments.index, dtype="float64")
    if overture.empty or segments.empty:
        return out
    ov, seg = _valid_gdf(overture), _valid_gdf(segments)
    tree = ov.sindex
    for seg_idx, geom in seg.geometry.items():
        if geom.is_empty or geom.area == 0:
            continue
        best, best_share = None, 0.0
        for pos in tree.query(geom, predicate="intersects"):
            share = ov.geometry.iloc[pos].intersection(geom).area / geom.area
            if share > best_share:
                best, best_share = ov.index[pos], share
        if best is not None and best_share >= min_overlap:
            out[seg_idx] = best
    return out


@dataclass
class MatchResult:
    """Hasil pencocokan: tabel per denah, penetapan segmen, dan segmen tanpa denah."""

    per_overture: pd.DataFrame
    assignment: pd.Series
    sam_only: pd.Index


def match(
    overture: gpd.GeoDataFrame,
    segments: gpd.GeoDataFrame,
    min_overlap: float = 0.5,
    match_iou: float = 0.5,
) -> MatchResult:
    """Cocokkan segmen SAM dengan denah Overture (many-to-one)."""
    ov, seg = _valid_gdf(overture), _valid_gdf(segments)
    assignment = assign_segments(ov, seg, min_overlap)
    rows = []
    for ov_idx, geom in ov.geometry.items():
        members = assignment.index[assignment == ov_idx]
        if len(members) == 0:
            rows.append((ov_idx, 0, 0.0, 0.0, False))
            continue
        union = unary_union(list(seg.geometry.loc[members]))
        value = iou(geom, union)
        rows.append((ov_idx, len(members), round(float(union.area), 2), value, value >= match_iou))
    per = pd.DataFrame(
        rows, columns=["overture_idx", "n_segments", "union_area_m2", "iou", "matched"]
    ).set_index("overture_idx")
    per = per.astype({"n_segments": int, "union_area_m2": float, "iou": float, "matched": bool})
    return MatchResult(per, assignment, assignment.index[assignment.isna()])


def comparison_summary(
    overture: gpd.GeoDataFrame, segments: gpd.GeoDataFrame, result: MatchResult
) -> dict:
    """Ringkasan perbandingan. Kategori denah: cocok / kurang_cocok / hanya_overture."""
    per = result.per_overture
    matched = per["matched"]
    with_seg = per["n_segments"] > 0
    ious = per.loc[matched, "iou"]
    a_ov = float(overture.geometry.area.sum()) if len(overture) else 0.0
    a_sam = float(segments.geometry.area.sum()) if len(segments) else 0.0
    return {
        "cocok": int(matched.sum()),
        "kurang_cocok": int((with_seg & ~matched).sum()),
        "hanya_overture": int((~with_seg).sum()),
        "hanya_sam": int(len(result.sam_only)),
        "median_iou": round(float(ious.median()), 3) if len(ious) else 0.0,
        "luas_overture_m2": round(a_ov, 1),
        "luas_sam_m2": round(a_sam, 1),
        "selisih_luas_pct": round(100 * (a_sam - a_ov) / a_ov, 1) if a_ov else 0.0,
    }


def centroid_offsets(overture: gpd.GeoDataFrame, guided: gpd.GeoDataFrame) -> pd.DataFrame:
    """Pergeseran pusat (meter, CRS metrik) SAM dipandu terhadap denah Overture.

    Hanya indeks yang ada di kedua tabel. Kolom: dx, dy, d (jarak).
    """
    common = overture.index.intersection(guided.index)
    oc = overture.geometry.loc[common].centroid
    gc = guided.geometry.loc[common].centroid
    out = pd.DataFrame({"dx": gc.x - oc.x, "dy": gc.y - oc.y}, index=common)
    out["d"] = np.hypot(out["dx"], out["dy"])
    return out


def offset_summary(offsets: pd.DataFrame) -> dict:
    """Ringkasan pergeseran. `bias_ratio` ~1: seragam satu arah; ~0: acak (heuristik)."""
    if offsets.empty:
        return {"median_geser_m": 0.0, "rata_dx_m": 0.0, "rata_dy_m": 0.0, "bias_ratio": 0.0}
    mean_vec = float(np.hypot(offsets["dx"].mean(), offsets["dy"].mean()))
    mean_d = float(offsets["d"].mean())
    return {
        "median_geser_m": round(float(offsets["d"].median()), 2),
        "rata_dx_m": round(float(offsets["dx"].mean()), 2),
        "rata_dy_m": round(float(offsets["dy"].mean()), 2),
        "bias_ratio": round(mean_vec / mean_d, 3) if mean_d else 0.0,
    }
```

```python
"""Penyaring kandidat atap hasil SAM (luas, bentuk, kecerahan, vegetasi).

SAM tidak tahu apa itu "atap", jadi pohon, jalan, dan area tanpa data ikut
tersegmentasi. Fungsi di sini murni (tanpa torch). Semua ambang dari config.
"""

from __future__ import annotations

import geopandas as gpd
import numpy as np
import pandas as pd
from rasterio import features
from rasterio.transform import Affine


def compute_exg(image: np.ndarray) -> np.ndarray:
    """Excess Green index (2g - r - b) pada koordinat kromatik; citra (3, H, W)."""
    rgb = image.astype("float32")
    total = rgb.sum(axis=0)
    total[total == 0] = 1.0
    r, g, b = rgb / total
    return (2 * g - r - b).astype("float32")


def zonal_mean(gdf: gpd.GeoDataFrame, band: np.ndarray, transform: Affine) -> pd.Series:
    """Rata-rata nilai `band` (H, W) di dalam tiap poligon; NaN jika di luar citra."""
    height, width = band.shape
    inv = ~transform
    values = []
    for geom in gdf.geometry:
        minx, miny, maxx, maxy = geom.bounds
        c_a, r_a = inv * (minx, miny)
        c_b, r_b = inv * (maxx, maxy)
        row0, row1 = int(max(0, np.floor(min(r_a, r_b)))), int(min(height, np.ceil(max(r_a, r_b))))
        col0, col1 = int(max(0, np.floor(min(c_a, c_b)))), int(min(width, np.ceil(max(c_a, c_b))))
        if row1 <= row0 or col1 <= col0:
            values.append(np.nan)
            continue
        sub = band[row0:row1, col0:col1]
        sub_t = transform * Affine.translation(col0, row0)
        mask = features.geometry_mask([geom], sub.shape, sub_t, invert=True)
        if not mask.any():
            mask = features.geometry_mask([geom], sub.shape, sub_t, invert=True, all_touched=True)
        values.append(float(sub[mask].mean()) if mask.any() else np.nan)
    return pd.Series(values, index=gdf.index, dtype="float64")


def apply_filters(
    gdf: gpd.GeoDataFrame, image: np.ndarray, transform: Affine, cfg: dict
) -> tuple[gpd.GeoDataFrame, dict]:
    """Terapkan penyaring berurutan; kembalikan (kandidat lolos, laporan jumlah per langkah)."""
    report = {"awal": len(gdf)}
    out = gdf
    if out.empty:
        return out, {**report, "lolos_luas": 0, "lolos_solidity": 0,
                     "lolos_kecerahan": 0, "lolos_vegetasi": 0}

    area = out.geometry.area
    out = out[(area >= cfg["min_area_m2"]) & (area <= cfg["max_area_m2"])]
    report["lolos_luas"] = len(out)

    hull = out.geometry.convex_hull.area
    out = out[(out.geometry.area / hull.where(hull > 0, np.nan)) >= cfg["min_solidity"]]
    report["lolos_solidity"] = len(out)

    brightness = image.astype("float32").mean(axis=0)
    out = out[zonal_mean(out, brightness, transform) >= cfg["min_brightness"]]
    report["lolos_kecerahan"] = len(out)

    out = out[zonal_mean(out, compute_exg(image), transform) <= cfg["max_exg"]]
    report["lolos_vegetasi"] = len(out)
    return out.reset_index(drop=True), report
```

```python
"""Utilitas raster dan poligon untuk deteksi atap (tanpa torch).

Menyiapkan citra kerja (di-clip ke AOI, di-reproject ke CRS metrik, diturunkan
resolusinya), membagi citra ke tile, dan mengubah mask SAM menjadi poligon.
"""

from __future__ import annotations

from pathlib import Path

import geopandas as gpd
import numpy as np
import rasterio
from rasterio import features
from rasterio.enums import Resampling
from rasterio.transform import Affine, from_origin
from rasterio.vrt import WarpedVRT
from shapely.geometry import shape
from shapely.geometry.base import BaseGeometry

from compare import iou


def prepare_working_image(
    ortho_path: Path, aoi_geom: BaseGeometry, aoi_crs, dst_crs, gsd_m: float, pad_m: float = 5.0
) -> tuple[np.ndarray, Affine]:
    """Baca ortofoto untuk bounding box AOI pada resolusi `gsd_m` di `dst_crs`.

    Mengembalikan citra (3, H, W) uint8 dan transform-nya. Di luar data valid = 0.
    """
    minx, miny, maxx, maxy = gpd.GeoSeries([aoi_geom], crs=aoi_crs).to_crs(dst_crs).total_bounds
    minx, miny, maxx, maxy = minx - pad_m, miny - pad_m, maxx + pad_m, maxy + pad_m
    width = int(np.ceil((maxx - minx) / gsd_m))
    height = int(np.ceil((maxy - miny) / gsd_m))
    transform = from_origin(minx, maxy, gsd_m, gsd_m)
    with rasterio.open(ortho_path) as src, WarpedVRT(
        src, crs=dst_crs, transform=transform, width=width, height=height,
        resampling=Resampling.average, nodata=0,
    ) as vrt:
        image = vrt.read([1, 2, 3])
    return image, transform


def tile_windows(height: int, width: int, tile: int, overlap: int) -> list[tuple[int, int, int, int]]:
    """Jendela (row0, col0, row1, col1) yang menutup seluruh citra dengan tumpang tindih."""
    step = tile - overlap

    def starts(n: int) -> list[int]:
        if n <= tile:
            return [0]
        s = list(range(0, n - tile, step))
        s.append(n - tile)
        return s

    return [(r, c, min(r + tile, height), min(c + tile, width)) for r in starts(height) for c in starts(width)]


def mask_to_polygons(mask: np.ndarray, transform: Affine) -> list:
    """Ubah mask boolean (H, W) menjadi poligon pada koordinat dunia."""
    return [
        shape(geom)
        for geom, value in features.shapes(mask.astype("uint8"), mask=mask, transform=transform)
        if value == 1
    ]


def dedupe_polygons(gdf: gpd.GeoDataFrame, iou_threshold: float = 0.7) -> gpd.GeoDataFrame:
    """Buang duplikat (IoU >= ambang), pertahankan yang lebih besar. Untuk zona overlap tile."""
    if gdf.empty:
        return gdf.reset_index(drop=True)
    order = gdf.geometry.area.sort_values(ascending=False).index
    sindex = gdf.sindex
    kept: list = []
    kept_set: set = set()
    for idx in order:
        geom = gdf.geometry.loc[idx]
        duplicate = False
        for pos in sindex.query(geom, predicate="intersects"):
            other = gdf.index[pos]
            if other in kept_set and iou(geom, gdf.geometry.loc[other]) >= iou_threshold:
                duplicate = True
                break
        if not duplicate:
            kept.append(idx)
            kept_set.add(idx)
    return gdf.loc[kept].reset_index(drop=True)


def clip_to_aoi(gdf: gpd.GeoDataFrame, aoi_geom: BaseGeometry) -> gpd.GeoDataFrame:
    """Potong poligon ke AOI; buang yang kosong; pecah multipart jadi Polygon tunggal."""
    out = gdf[gdf.intersects(aoi_geom)].copy()
    out["geometry"] = out.geometry.intersection(aoi_geom)
    out = out[~out.geometry.is_empty].explode(index_parts=False)
    return out[out.geom_type == "Polygon"].reset_index(drop=True)
```

```python
def resolve_device(requested: str, cuda_available: bool) -> str:
    """Pilih device. 'auto'/'cuda' tanpa CUDA jatuh ke CPU dengan peringatan."""
    if requested == "cpu":
        return "cpu"
    if requested in ("auto", "cuda"):
        if cuda_available:
            return "cuda"
        warnings.warn("CUDA tidak tersedia, memakai CPU (akan jauh lebih lambat).", UserWarning)
        return "cpu"
    raise ValueError(f"device tidak dikenal: {requested!r} (pakai auto, cuda, atau cpu)")


def ensure_checkpoint(model_type: str, directory: Path) -> Path:
    """Kembalikan path bobot; unduh jika belum ada."""
    if model_type not in CHECKPOINTS:
        raise ValueError(f"model_type {model_type!r} tidak dikenal; pilih salah satu {list(CHECKPOINTS)}")
    name, size = CHECKPOINTS[model_type]
    path = Path(directory) / name
    if not path.exists():
        path.parent.mkdir(parents=True, exist_ok=True)
        print(f"Mengunduh bobot SAM {model_type} ({size}) ke {path} ...")
        urllib.request.urlretrieve(BASE_URL + name, path)
    return path


class SamBackend:
    """SAM untuk segmentasi otomatis (`generate`) dan dipandu kotak (`predict_box`)."""

    def __init__(self, model_type: str, checkpoint_dir: Path, device: str = "auto", amg: dict | None = None):
        try:
            import torch
            from segment_anything import SamAutomaticMaskGenerator, SamPredictor, sam_model_registry
        except ImportError as exc:  # pragma: no cover - bergantung lingkungan
            raise SystemExit(
                "torch/segment-anything belum terpasang. Lihat README: pip install -r requirements-sam.txt"
            ) from exc
        self.device = resolve_device(device, torch.cuda.is_available())
        checkpoint = ensure_checkpoint(model_type, Path(checkpoint_dir))
        sam = sam_model_registry[model_type](checkpoint=str(checkpoint)).to(self.device)
        self._generator = SamAutomaticMaskGenerator(sam, **(amg or {}))
        self._predictor = SamPredictor(sam)

    def generate(self, image_hwc: np.ndarray) -> list[np.ndarray]:
        """Mask boolean (H, W) hasil segmentasi otomatis pada citra RGB uint8 (H, W, 3)."""
        return [m["segmentation"] for m in self._generator.generate(image_hwc)]

    def predict_box(self, image_hwc: np.ndarray, box_xyxy) -> tuple[np.ndarray, np.ndarray]:
        """Tiga kandidat mask (3, H, W) dan skornya untuk satu kotak (x0, y0, x1, y1) piksel."""
        self._predictor.set_image(image_hwc)
        masks, scores, _ = self._predictor.predict(
            box=np.asarray(box_xyxy, dtype="float32"), multimask_output=True
        )
        return masks, scores
```

```python
Semua path dan parameter dibaca dari config.yaml (bagian `sam` dan `compare`).

from __future__ import annotations

import sys
from pathlib import Path

sys.path.insert(0, str(Path(__file__).resolve().parent))

import geopandas as gpd  # noqa: E402
import numpy as np  # noqa: E402
from rasterio.transform import Affine  # noqa: E402

import roof_raster  # noqa: E402

def run_auto(backend, image: np.ndarray, transform: Affine, crs, sam_cfg: dict) -> gpd.GeoDataFrame:
    """SAM otomatis per tile -> poligon di CRS metrik, duplikat zona overlap dibuang."""
    _, height, width = image.shape
    polygons = []
    for row0, col0, row1, col1 in roof_raster.tile_windows(
        height, width, sam_cfg["tile_px"], sam_cfg["overlap_px"]
    ):
        tile = np.ascontiguousarray(image[:, row0:row1, col0:col1].transpose(1, 2, 0))
        tile_transform = transform * Affine.translation(col0, row0)
        for mask in backend.generate(tile):
            polygons.extend(roof_raster.mask_to_polygons(mask, tile_transform))
    segments = gpd.GeoDataFrame(geometry=polygons, crs=crs)
    return roof_raster.dedupe_polygons(segments, sam_cfg["auto"]["dedupe_iou"])


def write_geojson(gdf: gpd.GeoDataFrame, path: Path) -> None:
    """Tulis GeoJSON WGS84. GeoDataFrame kosong ditulis sebagai FeatureCollection kosong."""
    path.parent.mkdir(parents=True, exist_ok=True)
    if gdf.empty:
        path.write_text('{"type": "FeatureCollection", "features": []}', encoding="utf-8")
    else:
        gdf.to_crs("EPSG:4326").to_file(path, driver="GeoJSON")
```

```python
import compare  # noqa: E402


def run_guided(
    backend, image: np.ndarray, transform: Affine, crs, overture: gpd.GeoDataFrame, sam_cfg: dict
) -> gpd.GeoDataFrame:
    """SAM dipandu: tiap denah Overture jadi prompt kotak pada crop di sekitarnya.

    Dipilih satu dari tiga mask dengan IoU tertinggi terhadap denah. Denah yang
    tidak menghasilkan mask dilewati (tidak ada barisnya pada hasil).
    """
    _, height, width = image.shape
    inv = ~transform
    gsd = abs(transform.a)
    pad = sam_cfg["guided"]["box_pad_m"] / gsd
    rows = []
    for idx, geom in overture.geometry.items():
        minx, miny, maxx, maxy = geom.bounds
        c0, r1 = inv * (minx, miny)
        c1, r0 = inv * (maxx, maxy)
        bx0, by0, bx1, by1 = c0 - pad, r0 - pad, c1 + pad, r1 + pad
        margin = max(20.0, 0.5 * max(bx1 - bx0, by1 - by0))
        crop_r0, crop_r1 = max(0, int(by0 - margin)), min(height, int(np.ceil(by1 + margin)))
        crop_c0, crop_c1 = max(0, int(bx0 - margin)), min(width, int(np.ceil(bx1 + margin)))
        if crop_r1 <= crop_r0 or crop_c1 <= crop_c0:
            continue
        crop = np.ascontiguousarray(image[:, crop_r0:crop_r1, crop_c0:crop_c1].transpose(1, 2, 0))
        box_crop = [
            max(0.0, bx0 - crop_c0), max(0.0, by0 - crop_r0),
            min(crop.shape[1], bx1 - crop_c0), min(crop.shape[0], by1 - crop_r0),
        ]
        masks, scores = backend.predict_box(crop, box_crop)
        crop_transform = transform * Affine.translation(crop_c0, crop_r0)
        best = None
        for mask, score in zip(masks, scores):
            polys = roof_raster.mask_to_polygons(mask, crop_transform)
            if not polys:
                continue
            poly = max(polys, key=lambda p: p.area)
            value = compare.iou(geom, poly)
            if best is None or value > best[0]:
                best = (value, poly, float(score))
        if best is not None:
            rows.append({"idx": idx, "geometry": best[1], "iou_overture": round(best[0], 3),
                         "sam_score": round(best[2], 3), "area_m2": round(best[1].area, 2)})
    if not rows:
        return gpd.GeoDataFrame({"iou_overture": [], "sam_score": [], "area_m2": []}, geometry=[], crs=crs)
    out = gpd.GeoDataFrame(rows, geometry="geometry", crs=crs).set_index("idx")
    out.index.name = None
    return out
```
